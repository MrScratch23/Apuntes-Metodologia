🚀 Arrays y Bucles en JavaScript

Guía completa de métodos y bucles con ejemplos prácticos

🔄

Métodos Mutadores

push()
Agrega elementos al final del array
let arr = [1, 2, 3];
arr.push(4, 5); // Resultado: [1, 2, 3, 4, 5]
pop()
Elimina el último elemento
let arr = [1, 2, 3];
arr.pop(); // Resultado: [1, 2]
shift()
Elimina el primer elemento
let arr = [1, 2, 3];
arr.shift(); // Resultado: [2, 3]
unshift()
Agrega elementos al inicio
let arr = [1, 2, 3];
arr.unshift(0); // Resultado: [0, 1, 2, 3]
splice()
Agrega/elimina elementos
let arr = [1, 2, 3, 4, 5];
arr.splice(1, 2, 'a', 'b'); // Resultado: [1, 'a', 'b', 4, 5]
reverse()
Invierte el orden
let arr = [1, 2, 3];
arr.reverse(); // Resultado: [3, 2, 1]
sort()
Ordena elementos
let arr = [40, 1, 5, 200];
arr.sort(function(a, b) { return a - b; }); // [1, 5, 40, 200]
🔍

Métodos de Búsqueda

find()
Encuentra primer elemento que cumpla condición
let arr = [1, 2, 3, 4, 5];
arr.find(function(x) { return x > 2; }); // Resultado: 3
filter()
Filtra elementos que cumplan condición
let arr = [1, 2, 3, 4, 5, 6];
arr.filter(function(x) { return x % 2 === 0; }); // [2, 4, 6]
includes()
Verifica si contiene elemento
let arr = [1, 2, 3, 4, 5];
arr.includes(3); // Resultado: true
indexOf()
Encuentra índice de elemento
let arr = ['a', 'b', 'c', 'b'];
arr.indexOf('b'); // Resultado: 1
some()
Verifica si ALGÚN elemento cumple
let arr = [1, 2, 3, 4, 5];
arr.some(function(x) { return x > 3; }); // Resultado: true
every()
Verifica si TODOS cumplen
let arr = [1, 2, 3, 4, 5];
arr.every(function(x) { return x > 0; }); // Resultado: true
🔄

Métodos de Iteración

forEach()
Ejecuta función para cada elemento
let arr = [1, 2, 3];
arr.forEach(function(elemento) { console.log(elemento); }); // 1, 2, 3
map()
Transforma cada elemento
let arr = [1, 2, 3];
arr.map(x => x * 2); // Resultado: [2, 4, 6]
reduce()
Reduce a un solo valor
let arr = [1, 2, 3, 4, 5];
arr.reduce(function(acc, val) { return acc + val; }, 0); // Resultado: 15
flat()
Aplana arrays anidados
let arr = [1, [2, 3], [4, [5, 6]]];
arr.flat(2); // Resultado: [1, 2, 3, 4, 5, 6]
🛠️

Métodos de Utilidad

concat()
Une arrays
let arr1 = [1, 2]; let arr2 = [3, 4];
arr1.concat(arr2); // Resultado: [1, 2, 3, 4]
join()
Une elementos en string
let arr = ['Hola', 'mundo'];
arr.join(' '); // Resultado: "Hola mundo"
slice()
Extrae parte del array
let arr = [1, 2, 3, 4, 5];
arr.slice(1, 4); // Resultado: [2, 3, 4]
Array.isArray()
Verifica si es array
let arr = [1, 2, 3];
Array.isArray(arr); // Resultado: true

Bucles Tradicionales

for clásico
Bucle tradicional con contador
let arr = ['a', 'b', 'c'];
for (let i = 0; i < arr.length; i++) { console.log(arr[i]); } // Resultado: 'a', 'b', 'c'
for...in
Itera sobre propiedades enumerables
let arr = ['x', 'y', 'z'];
for (let indice in arr) { console.log(indice + ': ' + arr[indice]); } // Resultado: '0: x', '1: y', '2: z'
for...of
Itera sobre valores de objetos iterables
let arr = [10, 20, 30];
for (let valor of arr) { console.log(valor); } // Resultado: 10, 20, 30
while
Ejecuta mientras la condición sea true
let i = 0;
while (i < 3) { console.log(i); i++; } // Resultado: 0, 1, 2
do...while
Ejecuta al menos una vez
let i = 0;
do { console.log(i); i++; } while (i < 3); // Resultado: 0, 1, 2
🎯

Bucles Modernos con Arrays

forEach()
Método específico para arrays
let frutas = ['manzana', 'naranja', 'plátano'];
frutas.forEach(function(fruta, indice) { console.log(indice + ': ' + fruta); }); // Resultado: '0: manzana', '1: naranja', '2: plátano'
for...of con entries()
Obtiene índice y valor
let colores = ['rojo', 'verde', 'azul'];
for (let [indice, color] of colores.entries()) { console.log(indice + ': ' + color); } // Resultado: '0: rojo', '1: verde', '2: azul'
for...of con break
Puede romper el bucle
let numeros = [1, 2, 3, 4, 5];
for (let num of numeros) { if (num > 3) break; console.log(num); } // Resultado: 1, 2, 3
for...of con continue
Puede saltar iteraciones
let numeros = [1, 2, 3, 4, 5];
for (let num of numeros) { if (num % 2 === 0) continue; console.log(num); } // Resultado: 1, 3, 5
