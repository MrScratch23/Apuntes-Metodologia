🐘 Arrays en PHP

Guía completa de funciones para trabajar con arrays

🆕

Creación y Construcción

array() o []
Crear un nuevo array
-
$frutas = array("manzana", "naranja", "plátano"); $numeros = [1, 2, 3, 4, 5];
range()
Crear array con rango de valores
-
$numeros = range(1, 5); // Resultado: [1, 2, 3, 4, 5] $letras = range('a', 'd'); // Resultado: ['a', 'b', 'c', 'd']
array_fill()
Llenar array con valor específico
-
$array = array_fill(0, 3, 'valor'); // Resultado: [0 => 'valor', 1 => 'valor', 2 => 'valor']
array_combine()
Combinar dos arrays en uno asociativo
$claves = ['a', 'b', 'c']; $valores = [1, 2, 3];
$resultado = array_combine($claves, $valores); // Resultado: ['a' => 1, 'b' => 2, 'c' => 3]
🔧

Modificación

array_push()
Agregar elementos al final
$array = [1, 2, 3];
array_push($array, 4, 5); // Resultado: [1, 2, 3, 4, 5]
array_pop()
Eliminar último elemento
$array = [1, 2, 3];
$ultimo = array_pop($array); // $ultimo = 3, $array = [1, 2]
array_shift()
Eliminar primer elemento
$array = [1, 2, 3];
$primero = array_shift($array); // $primero = 1, $array = [2, 3]
array_unshift()
Agregar elementos al inicio
$array = [2, 3];
array_unshift($array, 0, 1); // Resultado: [0, 1, 2, 3]
array_splice()
Eliminar y reemplazar elementos
$array = [1, 2, 3, 4, 5];
array_splice($array, 1, 2, ['a', 'b']); // Resultado: [1, 'a', 'b', 4, 5]
🔍

Búsqueda y Filtrado

in_array()
Verificar si valor existe en array
$array = [1, 2, 3, 4, 5];
in_array(3, $array); // true in_array(10, $array); // false
array_search()
Buscar valor y devolver clave
$array = ['a' => 1, 'b' => 2, 'c' => 3];
array_search(2, $array); // 'b' array_search(5, $array); // false
array_filter()
Filtrar elementos con callback
$numeros = [1, 2, 3, 4, 5, 6];
$pares = array_filter($numeros, function($n) { return $n % 2 == 0; }); // Resultado: [2, 4, 6]
array_keys()
Obtener todas las claves
$array = ['a' => 1, 'b' => 2, 'c' => 3];
array_keys($array); // Resultado: ['a', 'b', 'c']
array_values()
Obtener todos los valores
$array = ['a' => 1, 'b' => 2, 'c' => 3];
array_values($array); // Resultado: [1, 2, 3]
🔄

Iteración y Transformación

array_map()
Aplicar función a cada elemento
$numeros = [1, 2, 3, 4];
$duplicados = array_map(function($n) { return $n * 2; }, $numeros); // Resultado: [2, 4, 6, 8]
array_walk()
Iterar sobre elementos (modifica original)
$array = ['a' => 1, 'b' => 2, 'c' => 3];
array_walk($array, function(&$valor, $clave) { $valor = $clave . ':' . $valor; }); // Resultado: ['a' => 'a:1', 'b' => 'b:2', 'c' => 'c:3']
array_reduce()
Reducir array a un solo valor
$numeros = [1, 2, 3, 4, 5];
$suma = array_reduce($numeros, function($acum, $num) { return $acum + $num; }, 0); // Resultado: 15
foreach
Bucle específico para arrays
$usuarios = ['Ana', 'Juan', 'Maria'];
foreach ($usuarios as $usuario) { echo "Usuario: $usuario\n"; } // Resultado: Usuario: Ana, Usuario: Juan, Usuario: Maria
📊

Ordenamiento

sort()
Ordenar valores (ascendente)
$array = [3, 1, 4, 2];
sort($array); // Resultado: [1, 2, 3, 4]
rsort()
Ordenar valores (descendente)
$array = [3, 1, 4, 2];
rsort($array); // Resultado: [4, 3, 2, 1]
asort()
Ordenar manteniendo claves
$array = ['b' => 2, 'a' => 1, 'c' => 3];
asort($array); // Resultado: ['a' => 1, 'b' => 2, 'c' => 3]
ksort()
Ordenar por claves
$array = ['z' => 1, 'a' => 2, 'm' => 3];
ksort($array); // Resultado: ['a' => 2, 'm' => 3, 'z' => 1]
usort()
Ordenar con función personalizada
$array = [5, 2, 8, 1];
usort($array, function($a, $b) { return $a - $b; }); // Resultado: [1, 2, 5, 8]
🛠️

Utilidades y Combinación

array_merge()
Combinar arrays
$array1 = [1, 2]; $array2 = [3, 4];
$combinado = array_merge($array1, $array2); // Resultado: [1, 2, 3, 4]
array_slice()
Extraer parte del array
$array = [1, 2, 3, 4, 5];
$parte = array_slice($array, 1, 3); // Resultado: [2, 3, 4]
array_chunk()
Dividir array en fragmentos
$array = [1, 2, 3, 4, 5, 6];
$trozos = array_chunk($array, 2); // Resultado: [[1, 2], [3, 4], [5, 6]]
array_unique()
Eliminar valores duplicados
$array = [1, 2, 2, 3, 3, 3];
$unicos = array_unique($array); // Resultado: [1, 2, 3]
count()
Contar elementos
$array = [1, 2, 3, 4, 5];
count($array); // 5
implode()
Unir elementos en string
$array = ['Hola', 'mundo', 'PHP'];
implode(' ', $array); // Resultado: "Hola mundo PHP"
explode()
Dividir string en array
$texto = "manzana,naranja,plátano";
explode(',', $texto); // Resultado: ['manzana', 'naranja', 'plátano']
